document.addEventListener('DOMContentLoaded', function() {
  const toggleFaqBtn = document.getElementById('toggleFaqViewBtn');
  const faqQuestionsBlock = document.getElementById('faqQuestionsBlock');
  const faqTemplateBlock = document.getElementById('faqTemplateFields');

  toggleFaqBtn.addEventListener('click', () => {
    if (faqTemplateBlock.classList.contains('hidden-fields')) {
      faqTemplateBlock.classList.remove('hidden-fields');
      faqQuestionsBlock.classList.add('hidden-fields');
      toggleFaqBtn.textContent = 'Use Individual Questions';
    } else {
      faqTemplateBlock.classList.add('hidden-fields');
      faqQuestionsBlock.classList.remove('hidden-fields');
      toggleFaqBtn.textContent = 'Use Title, Summary & Keywords Template';
    }
  });
});
